cmake_minimum_required(VERSION 3.22.1)
project(minepilot_bridge LANGUAGES C)
add_library(minepilot_bridge SHARED minepilot_bridge.c)
set_target_properties(minepilot_bridge PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)

if(DEFINED MINEPILOT_NODE_SMOKE_DIR)
    enable_language(CXX)
    add_library(node_mobile SHARED IMPORTED)
    set_target_properties(node_mobile PROPERTIES
        IMPORTED_LOCATION "${MINEPILOT_NODE_SMOKE_DIR}/bin/${ANDROID_ABI}/libnode.so")
    add_library(minepilot_node_smoke SHARED node_smoke.cpp)
    target_compile_features(minepilot_node_smoke PRIVATE cxx_std_20)
    target_include_directories(minepilot_node_smoke SYSTEM PRIVATE "${MINEPILOT_NODE_SMOKE_DIR}/include/node")
    target_link_libraries(minepilot_node_smoke PRIVATE node_mobile)
endif()
